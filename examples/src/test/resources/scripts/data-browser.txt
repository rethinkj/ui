# The data browser: a million made-up orders, filtered as one types, the grid's keys, the sort
# by id and a resized column.
size 900 520
frame
snap browser-top                 # a million orders, none selected

press #filter
type "ada paid"                  # each word in some text column: every twentieth order
frame
advance 1                        # the field's border turns to the accent colour
frame
snap browser-filtered

key Tab                          # on to the grid, with a ring
key ArrowDown                    # the cursor on the first order shown
key End control                  # and on the last cell of the last one
frame
snap browser-end

press #filter
key A control
key Backspace                    # no filter: the order stays selected, and the grid shows it
frame
advance 1
frame
snap browser-unfiltered

press #column-0                  # sorts by id, descending: the order is near the top now
frame
wheel #rows -2000 0              # the columns back to their start; the cursor stays
frame
down #resize-2                   # the right edge of email, at 499,86
frame
move 559,86                      # 60 pixels to the right, a frame for each step as in a window
frame
up 559,86
frame
snap browser-sorted

# Editing: changes collect until Commit, with their marks, and filters per column
key Home control                 # the first cell: the id of the last order, which does not edit
key ArrowRight                   # customer
frame
type "Ada King"                  # typing replaces the value
key Enter                        # handed on, a row down: the row is changed
key ArrowRight
key ArrowRight
key ArrowRight                   # country
frame
type "IS"
key Tab                          # handed on, a column on
key Insert                       # a new order after this one, its number a +
key Home
key ArrowRight                   # its customer
frame
type "Grace Hopper"
key Enter
key ArrowDown
key Delete control               # an order marked to be deleted
frame
snap browser-changes

key End
key ArrowLeft                    # status
key ArrowDown alt                # its filter: the statuses to keep
frame
advance 1
frame
press #value-refunded
press #value-pending
press label="Apply"
frame
advance 1
frame
snap browser-filters

press role=switch label="Dark"   # the light theme, the changes and filters as they were
frame
advance 1
frame
snap browser-light

press #commit                    # written: the marks go, the deleted order with them
frame
snap browser-committed
