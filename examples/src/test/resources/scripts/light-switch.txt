# The light switch example: pressed on, then off again with Tab and Space.
size 360 240 scale 2
frame
snap light-switch-off

press role=switch label="Lamp"
frame
advance 0.1
frame
snap light-switch-moving

advance 1
frame
snap light-switch-on

key Tab                # the press left the focus alone; Tab gives it to the switch, with a ring
key Space
frame
advance 1
frame
snap light-switch-focused-off
