# The notes of the example (Notes): reordered in the row, torn off into a window that follows the
# pointer, edited there, docked back where the marker shows, and closed back into the row.
size 640 300 at 100,100
frame
snap notes-row

down #handle-Ideas         # Ideas along the row, behind Todo
move 220,80
move 560,80
frame
up 560,80
frame
advance 1                  # the cards spring to their places
frame

down #handle-Groceries     # Groceries out of the window, above it
move 60,80
move 60,-60
frame
advance 1                  # the card in the row turns muted
frame
snap notes-torn            # its own window under the pointer, held at its title
move screen 800,220        # the window follows
frame
up screen 800,220
frame

window Groceries           # an edit in the note's window shows in the row at once
press #text
key End control
type ", milk"
frame
snap notes-edited

window main
down #handle-Todo          # Todo out of the window, and back at the start of the row
move 260,80
move 260,-60
frame
move screen 140,200
frame
advance 1
frame
up screen 140,200
frame
advance 1
frame

window Groceries           # closing the note's window docks it where its muted card is
close
frame
advance 1
frame
snap notes-closed
